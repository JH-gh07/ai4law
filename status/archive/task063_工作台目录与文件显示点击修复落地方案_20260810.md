### 1.4 后端: 输入文件预览/下载 API

**问题**: 当前 `/api/v1/artifacts/preview` 的 `_resolve_artifact_path` 和 `_assert_artifact_access` 假定文件路径在 `outputs/` 目录下，通过 `ReportArtifactModel` 或 `UploadedFileModel` 关联用户。

**方案**: 利用已有的 `UploadedFileModel`（`backend/models/review.py`），其 `storage_path` 字段存储上传文件路径。扩展 artifact API 以支持输入文件查询。

**文件**: `backend/api/v1/endpoints/artifacts.py`

新增查询参数 `source`，默认 `"output"`，可选 `"input"`（或 `"upload"`）：

```python
@router.get("/preview", response_model=ArtifactPreviewResponse)
def preview_artifact(
    path: str = Query(..., min_length=1),
    source: str = Query("output", description="'output' for generated artifacts, 'input' for uploaded files"),
    # ... 其余参数不变
):
    if source == "input":
        return _preview_input_file(path, db, current_user, container)
    return _preview_output_artifact(path, db, current_user, container)
```

**新函数 `_preview_input_file`**:

```python
def _preview_input_file(
    raw_path: str,
    db: Session,
    current_user: AuthUser,
    container,
) -> ArtifactPreviewResponse:
    """预览用户上传的输入文件。

    与 _preview_output_artifact 的区别：
    - 路径解析范围包括 uploads/ 目录
    - 权限通过 UploadedFileModel 而非 ReportArtifactModel 检查
    """
    candidate = Path(raw_path)
    resolved = candidate.resolve() if candidate.is_absolute() else (Path.cwd() / candidate).resolve()

    # 扩允输入文件根目录
    allowed_roots = [
        container.settings.upload_dir.resolve(),
        container.settings.storage_dir.resolve(),
        Path.cwd() / "benchmarks",
        Path.cwd() / "resources",
    ]
    # 回退到 _allowed_roots 以支持混合场景
    for root in allowed_roots:
        try:
            resolved.relative_to(root)
            break
        except ValueError:
            continue
    else:
        # 尝试通用 allowed_roots
        for root in _allowed_roots(container):
            try:
                resolved.relative_to(root)
                break
            except ValueError:
                continue
        else:
            raise HTTPException(status_code=403, detail="Input file path is outside allowed scope.")

    if not resolved.exists() or not resolved.is_file():
        raise HTTPException(status_code=404, detail="Input file not found.")

    # 权限：通过 UploadedFileModel 或 ReportArtifactModel 双重检查
    candidate_paths = tuple(_candidate_artifact_paths(resolved))
    upload_stmt = select(UploadedFileModel.id).where(
        UploadedFileModel.user_id == current_user.id,
        UploadedFileModel.storage_path.in_(candidate_paths),
    )
    report_stmt = select(ReportArtifactModel.id).where(
        ReportArtifactModel.user_id == current_user.id,
        ReportArtifactModel.file_path.in_(candidate_paths),
    )
    upload_hit = db.execute(upload_stmt).scalar_one_or_none()
    report_hit = db.execute(report_stmt).scalar_one_or_none()
    if not upload_hit and not report_hit:
        raise HTTPException(status_code=403, detail="You do not have access to this input file.")

    # 渲染逻辑与输出文件相同
    suffix = resolved.suffix.lower()
    parser = FileParser()
    file_url = f"/api/v1/artifacts/file?path={quote(str(resolved))}&source=input"

    if suffix == ".html":
        raw_html = resolved.read_text(encoding="utf-8", errors="ignore")
        return ArtifactPreviewResponse(
            path=str(resolved), file_name=resolved.name,
            kind=suffix.lstrip("."), render_mode="html",
            content=raw_html, file_url=file_url,
        )
    if suffix == ".pdf":
        return ArtifactPreviewResponse(
            path=str(resolved), file_name=resolved.name,
            kind="pdf", render_mode="pdf",
            content=parser.parse_text(str(resolved)),
            file_url=file_url,
        )
    if suffix in {".docx", ".md", ".txt", ".json", ".csv"}:
        return ArtifactPreviewResponse(
            path=str(resolved), file_name=resolved.name,
            kind=suffix.lstrip("."), render_mode="text",
            content=parser.parse_text(str(resolved)),
            file_url=file_url,
        )
    return ArtifactPreviewResponse(
        path=str(resolved), file_name=resolved.name,
        kind=suffix.lstrip(".") or "file", render_mode="download",
        content="", file_url=file_url,
    )
```

同样需要在 `/file` 和 `/download` 端点添加 `source` 参数支持。

// frontend/src/features/resource-explorer/output-artifacts.ts

export function buildOutputEntries(
  artifacts: OutputArtifact[], runs: ModuleRun[], lang: ResourceLanguage,
): OutputTreeEntry[] {
  const visibleArtifacts = filterUserFacingArtifacts(artifacts);

  // ✅ 新逻辑：按 runId 分组
  const runOrder = new Map(runs.map((run, index) => [run.id, index + 1]));

  const usedNames = new Map<string, number>();
  return visibleArtifacts.map((artifact) => {
    const runId = artifact.runId;
    const runNumber = runId ? runOrder.get(runId) ?? null : null;
    const runLabel = runNumber === null
      ? (lang === "zh" ? "未知批次生成结果" : "Unknown Run Results")
      : (lang === "zh" ? `第${runNumber}次生成结果` : `Run ${runNumber} Results`);
    const baseName = resolveArtifactDisplayName(artifact, lang);
    const key = `${runLabel}/${baseName}`;
    const count = (usedNames.get(key) ?? 0) + 1;
    usedNames.set(key, count);
    return {
      virtualPath: `${runLabel}/${count === 1 ? baseName : `${baseName} (${count})`}`,
      artifact,
    };
  });
}

### 5.1 `inferLabelKey` 防误匹配

**文件**: `frontend/src/features/resource-explorer/input-resources.ts` (line 16-26)

在正则匹配后增加扩展名检查，避免 `dpia_report.pdf`（输出报告）被误标为 DPIA 材料：

```typescript
const inferLabelKey = (path: string, hint?: string): string | undefined => {
  const normalizedHint = hint?.trim().toLowerCase();
  const exactHint = Object.keys(INPUT_ROLE_LABELS).find((key) => key === normalizedHint);
  if (exactHint) return exactHint;

  // ✅ 新增：有明确 hint（来自 file_role 字段）时才推断
  //   文件路径名匹配仅作为 hint 不存在时的 fallback
  if (hint) return undefined;  // hint 在 collectUserInputFiles 中已从 file_role 提取

  const normalizedPath = path.replace(/\\/g, "/").toLowerCase();
  return Object.keys(INPUT_ROLE_LABELS).find((key) => {
    const escapedKey = key.replace(/[.*+?^${}()|[\]\\]/g, "\\$&");
    return new RegExp(`(?:^|[\\/_.-])${escapedKey}(?:[\\/_.-]|$)`).test(normalizedPath);
  });
};
```

**注**: 此改动修复了误匹配，但也使得没有 `file_role` 的文件不再通过文件名推断标签。如果需要保留推断行为，可以增加黑名单（如 `report`、`output` 等关键词不触发推断）。
